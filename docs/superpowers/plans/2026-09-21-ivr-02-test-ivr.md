# Constructor reserves answer as pending, initial stage answering.
# challenge defaults to override or new_challenge(), and is validated if injected.
Flow(settings: Settings, call_control_id: str, call_leg_id: str,
     *, now: float, challenge: str | None = None)

# Public fields used by runtime/tests:
# stage: str, attempt: int, pending: Command | None, deadline: float,

Flow.expire(*, now: float) -> Command | None
Flow.command_failed(command_id: str, *, now: float) -> Command | None
